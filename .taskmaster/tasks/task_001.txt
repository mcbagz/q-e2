## 2. Create checkpoint_manager.f90 skeleton with global flags and public API stubs [done]

### Details:
• MODULE checkpoint_manager  
    USE ISO_C_BINDING  
    PRIVATE  
    PUBLIC :: install_signal_handlers, write_checkpoint, load_checkpoint, write_status_snapshot, soft_pause_requested, snapshot_requested  
    LOGICAL, VOLATILE :: soft_pause_requested = .FALSE., snapshot_requested = .FALSE.

       SUBROUTINE c_set_usr1_handler(f) BIND(C,NAME="c_set_usr1_handler")

       END SUBROUTINE  
       SUBROUTINE c_set_usr2_handler(f) BIND(C,NAME="c_set_usr2_handler")  
         IMPORT :: C_FUNPTR  
         TYPE(C_FUNPTR), VALUE :: f  
       END SUBROUTINE  
    END INTERFACE  
    CONTAINS  
      SUBROUTINE install_signal_handlers() ; END  
      SUBROUTINE write_checkpoint(fname,iter,etot,conv) ; END  
      SUBROUTINE load_checkpoint(fname) ; END

 END MODULE checkpoint_manager
